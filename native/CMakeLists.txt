cmake_minimum_required(VERSION 3.16)
project(radiant LANGUAGES CXX)
option(RADIANT_MPI "Enable MPI ranks alongside OpenMP threads" OFF)
find_package(OpenMP REQUIRED)
add_executable(radiant radiant.cpp)
target_compile_features(radiant PRIVATE cxx_std_11)
target_link_libraries(radiant PRIVATE OpenMP::OpenMP_CXX)
if(WIN32)
 target_link_libraries(radiant PRIVATE psapi)
endif()
if(RADIANT_MPI)
 find_package(MPI REQUIRED COMPONENTS CXX)
 target_compile_definitions(radiant PRIVATE RADIANT_MPI)
 target_link_libraries(radiant PRIVATE MPI::MPI_CXX)
endif()
